    function ie_admin() {
      return request.auth != null
        && request.auth.token.email == "dreamdosima08@gmail.com"
        && request.auth.token.email_verified == true;
    }
    function ie_giris() {
      return request.auth != null;
    }

    // Özel ilan kaydı (onay bekleyenler dahil). Sadece sahibi ve yönetici okur.
    match /ikinci_el_ilan/{id} {
      allow read: if ie_giris() && (resource.data.uid == request.auth.uid || ie_admin());
      allow create: if ie_giris()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.keys().hasOnly(['uid','ad','baslik','kategori','fiyat','pazarlik','hal','ilce','aciklama','kapak','oran','fotoSayisi','durum','zaman'])
        && request.resource.data.durum == 'bekliyor'
        && request.resource.data.zaman == request.time
        && request.resource.data.baslik is string && request.resource.data.baslik.size() >= 5 && request.resource.data.baslik.size() <= 80
        && request.resource.data.aciklama is string && request.resource.data.aciklama.size() >= 10 && request.resource.data.aciklama.size() <= 1500
        && request.resource.data.ad is string && request.resource.data.ad.size() <= 40
        && request.resource.data.kategori is string && request.resource.data.kategori.size() <= 20
        && request.resource.data.hal is string && request.resource.data.hal.size() <= 20
        && request.resource.data.ilce is string && request.resource.data.ilce.size() <= 20
        && request.resource.data.pazarlik is bool
        && request.resource.data.fiyat is number && request.resource.data.fiyat >= 1 && request.resource.data.fiyat <= 10000000
        && request.resource.data.oran is number && request.resource.data.oran > 0.2 && request.resource.data.oran < 5
        && request.resource.data.kapak is string && request.resource.data.kapak.size() <= 70000
        && request.resource.data.fotoSayisi is int && request.resource.data.fotoSayisi >= 1 && request.resource.data.fotoSayisi <= 3;
      // Sahibi sadece durumu "satildi" veya "kaldirildi" yapabilir. Yönetici her şeyi güncelleyebilir.
      allow update: if ie_admin()
        || (ie_giris() && resource.data.uid == request.auth.uid
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['durum'])
            && request.resource.data.durum in ['satildi','kaldirildi']);
      allow delete: if ie_admin();
    }

    // Herkese açık yayın kopyası (yönetici onaylayınca yazılır).
    match /ikinci_el_yayin/{id} {
      allow read: if true;
      allow create, delete: if ie_admin();
      allow update: if ie_admin()
        || (ie_giris() && resource.data.uid == request.auth.uid
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['satildi'])
            && request.resource.data.satildi == true);
      allow delete: if ie_admin() || (ie_giris() && resource.data.uid == request.auth.uid);
    }

    // Tam boy fotoğraflar. Tek tek okunabilir (tarama/liste yok).
    match /ikinci_el_foto/{id} {
      allow get: if true;
      allow list: if ie_admin();
      allow create: if ie_giris()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.keys().hasOnly(['uid','ilan','veri','zaman'])
        && request.resource.data.veri is string && request.resource.data.veri.size() <= 450000
        && request.resource.data.veri.matches('^data:image/jpeg;base64,.*')
        && request.resource.data.ilan is string && request.resource.data.ilan.size() <= 40
        && request.resource.data.zaman == request.time;
      allow delete: if ie_giris() && (resource == null || resource.data.uid == request.auth.uid || ie_admin());
    }

    // Telefon numarası. Sadece giriş yapmış kullanıcı tek tek okuyabilir; liste sadece yönetici.
    match /ikinci_el_iletisim/{id} {
      allow get: if ie_giris();
      allow list: if ie_admin();
      allow create: if ie_giris()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.keys().hasOnly(['uid','tel'])
        && request.resource.data.tel is string
        && request.resource.data.tel.matches('^05[0-9]{9}$');
      allow delete: if ie_giris() && (resource == null || resource.data.uid == request.auth.uid || ie_admin());
    }

    // Şikayetler. Giriş yapan yazar, sadece yönetici okur.
    match /ikinci_el_bildirim/{id} {
      allow create: if ie_giris()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.keys().hasOnly(['uid','ilan','baslik','neden','zaman'])
        && request.resource.data.neden is string && request.resource.data.neden.size() <= 300
        && request.resource.data.baslik is string && request.resource.data.baslik.size() <= 80
        && request.resource.data.ilan is string && request.resource.data.ilan.size() <= 40
        && request.resource.data.zaman == request.time;
      allow read, delete: if ie_admin();
    }
